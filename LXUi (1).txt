-- ─── Utility ─────────────────────────────────────────────────────────────────
local function Tween(obj, props, t, style, dir)
    local tw = TweenService:Create(obj,
        TweenInfo.new(t or 0.22, style or Enum.EasingStyle.Quart, dir or Enum.EasingDirection.Out),
        props)
    tw:Play()
    return tw
end

local function New(class, props)
    local o = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then o[k] = v end
    end
    if props and props.Parent then o.Parent = props.Parent end
    return o
end

local function Corner(p, r) return New("UICorner",{CornerRadius=UDim.new(0,r or 8),Parent=p}) end
local function Stroke(p, c, th, tr) return New("UIStroke",{Color=c or Color3.new(1,1,1),Thickness=th or 1,Transparency=tr or 0.8,Parent=p}) end
local function Pad(p, t, r, b, l) return New("UIPadding",{PaddingTop=UDim.new(0,t or 8),PaddingRight=UDim.new(0,r or 8),PaddingBottom=UDim.new(0,b or 8),PaddingLeft=UDim.new(0,l or 8),Parent=p}) end
local function List(p, gap, dir, ha, va)
    local l = New("UIListLayout",{Padding=UDim.new(0,gap or 6),FillDirection=dir or Enum.FillDirection.Vertical,HorizontalAlignment=ha or Enum.HorizontalAlignment.Left,VerticalAlignment=va or Enum.VerticalAlignment.Top,SortOrder=Enum.SortOrder.LayoutOrder,Parent=p})
    return l
end

local function Ripple(btn, color)
    local r = New("Frame",{
        Size=UDim2.new(0,0,0,0), AnchorPoint=Vector2.new(0.5,0.5),
        BackgroundColor3=color or Color3.fromRGB(255,255,255),
        BackgroundTransparency=0.55, BorderSizePixel=0,
        ClipsDescendants=false, ZIndex=btn.ZIndex+2, Parent=btn
    })
    Corner(r,999)
    local mp = UserInputService:GetMouseLocation()
    local bp = btn.AbsolutePosition
    r.Position = UDim2.new(0, mp.X-bp.X, 0, mp.Y-bp.Y)
    local sz = math.max(btn.AbsoluteSize.X, btn.AbsoluteSize.Y)*2.5
    Tween(r,{Size=UDim2.new(0,sz,0,sz),BackgroundTransparency=1},0.45,Enum.EasingStyle.Quad)
    task.delay(0.5, function() r:Destroy() end)
end

local function MakeDraggable(frame, handle)
    handle = handle or frame
    local drag, ds, sp = false, nil, nil
    handle.InputBegan:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
            drag=true; ds=i.Position; sp=frame.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if drag and (i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch) then
            local d=i.Position-ds
            frame.Position=UDim2.new(sp.X.Scale,sp.X.Offset+d.X,sp.Y.Scale,sp.Y.Offset+d.Y)
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
            drag=false
        end
    end)
end

                local F=New("Frame",{Size=UDim2.new(1,0,0,0),AutomaticSize=Enum.AutomaticSize.Y,BackgroundColor3=T.BgElem,ZIndex=10,Parent=container})
                Corner(F,7); Pad(F,10,12,10,12); List(F,4)
                if PT~="" then
                    New("TextLabel",{Size=UDim2.new(1,0,0,0),AutomaticSize=Enum.AutomaticSize.Y,BackgroundTransparency=1,Text=PT,TextColor3=T.Text,TextSize=12,Font=Enum.Font.GothamBold,TextXAlignment=Enum.TextXAlignment.Left,TextWrapped=true,ZIndex=11,Parent=F})
                end
                local CL=New("TextLabel",{Size=UDim2.new(1,0,0,0),AutomaticSize=Enum.AutomaticSize.Y,BackgroundTransparency=1,Text=PC,TextColor3=T.Sub,TextSize=11,Font=Enum.Font.Gotham,TextXAlignment=Enum.TextXAlignment.Left,TextWrapped=true,LineHeight=1.4,ZIndex=11,Parent=F})
                local o={}
                function o:Set(t,c) CL.Text=c or PC end
                return o
            end

        end -- MakeComponents

        MakeComponents(Tab, Page)
        return Tab
    end -- CreateTab

    -- ── Notify ────────────────────────────────────────────────────────────────
    function Window:Notify(cfg)
        cfg=cfg or {}
        local NT=cfg.Title or ""; local NC=cfg.Content or ""
        local ND=cfg.Duration or 4; local NType=cfg.Type or "Info"
        local NColor=T[NType] or T.Info

        local NF=New("Frame",{
            Size=UDim2.new(1,0,0,0),AutomaticSize=Enum.AutomaticSize.Y,
            BackgroundColor3=T.BgSecond,BackgroundTransparency=1,
            ZIndex=51,Parent=NotifHolder
        })
        Corner(NF,9); Stroke(NF,NColor,1,0.6)
        Pad(NF,10,12,10,14); List(NF,3)

        New("Frame",{Position=UDim2.new(0,0,0,0),Size=UDim2.new(0,3,1,0),BackgroundColor3=NColor,ZIndex=52,Parent=NF})
        New("TextLabel",{Size=UDim2.new(1,0,0,14),BackgroundTransparency=1,Text=NT,TextColor3=T.Text,TextSize=12,Font=Enum.Font.GothamBold,TextXAlignment=Enum.TextXAlignment.Left,ZIndex=52,Parent=NF})
        if NC~="" then
            New("TextLabel",{Size=UDim2.new(1,0,0,0),AutomaticSize=Enum.AutomaticSize.Y,BackgroundTransparency=1,Text=NC,TextColor3=T.Sub,TextSize=11,Font=Enum.Font.Gotham,TextXAlignment=Enum.TextXAlignment.Left,TextWrapped=true,ZIndex=52,Parent=NF})
        end
        local PBg=New("Frame",{Size=UDim2.new(1,0,0,2),BackgroundColor3=T.Border,ZIndex=52,Parent=NF})
        Corner(PBg,2)
        local PF=New("Frame",{Size=UDim2.new(1,0,1,0),BackgroundColor3=NColor,ZIndex=53,Parent=PBg})
        Corner(PF,2)

        NF.Position=UDim2.new(0.08,0,0,0)
        Tween(NF,{BackgroundTransparency=0.1,Position=UDim2.new(0,0,0,0)},0.3,Enum.EasingStyle.Back,Enum.EasingDirection.Out)
        Tween(PF,{Size=UDim2.new(0,0,1,0)},ND,Enum.EasingStyle.Linear)
        task.delay(ND,function()
            Tween(NF,{BackgroundTransparency=1,Position=UDim2.new(0.08,0,0,0)},0.25)
            task.delay(0.3,function() NF:Destroy() end)
        end)
    end

    -- ── Dialog ────────────────────────────────────────────────────────────────
    function Window:Dialog(cfg)
        cfg=cfg or {}
        local DTitle=cfg.Title or "Dialog"; local DContent=cfg.Content or ""
        local DBtns=cfg.Buttons or {{Name="OK",Callback=function()end}}

        local Dim=New("Frame",{Size=UDim2.new(1,0,1,0),BackgroundColor3=Color3.fromRGB(0,0,0),BackgroundTransparency=1,ZIndex=40,Parent=Gui})
        Tween(Dim,{BackgroundTransparency=0.55},0.22)

        local DF=New("Frame",{
            AnchorPoint=Vector2.new(0.5,0.5),Position=UDim2.new(0.5,0,0.5,0),
            Size=UDim2.new(0,320,0,80),AutomaticSize=Enum.AutomaticSize.Y,
            BackgroundColor3=T.BgSecond,ZIndex=41,Parent=Gui
        })
        Corner(DF,11); Stroke(DF,T.Border,1,0.5)
        Pad(DF,18,18,18,18); List(DF,10)
        Tween(DF,{Size=UDim2.new(0,320,0,0)},0.32,Enum.EasingStyle.Back,Enum.EasingDirection.Out)

        New("TextLabel",{Size=UDim2.new(1,0,0,18),BackgroundTransparency=1,Text=DTitle,TextColor3=T.Text,TextSize=14,Font=Enum.Font.GothamBold,TextXAlignment=Enum.TextXAlignment.Left,ZIndex=42,Parent=DF})
        if DContent~="" then
            New("TextLabel",{Size=UDim2.new(1,0,0,0),AutomaticSize=Enum.AutomaticSize.Y,BackgroundTransparency=1,Text=DContent,TextColor3=T.Sub,TextSize=12,Font=Enum.Font.Gotham,TextXAlignment=Enum.TextXAlignment.Left,TextWrapped=true,LineHeight=1.4,ZIndex=42,Parent=DF})
        end
        local BR=New("Frame",{Size=UDim2.new(1,0,0,30),BackgroundTransparency=1,ZIndex=42,Parent=DF})
        New("UIListLayout",{FillDirection=Enum.FillDirection.Horizontal,HorizontalAlignment=Enum.HorizontalAlignment.Right,VerticalAlignment=Enum.VerticalAlignment.Center,Padding=UDim.new(0,8),SortOrder=Enum.SortOrder.LayoutOrder,Parent=BR})

        local function CloseD()
            Tween(Dim,{BackgroundTransparency=1},0.2)
            Tween(DF,{BackgroundTransparency=1},0.2)
            task.delay(0.25,function() Dim:Destroy(); DF:Destroy() end)
        end
        for i,b in ipairs(DBtns) do
            local BC=b.Color or (i==1 and T.Accent or T.BgThird)
            local B=New("TextButton",{
                Size=UDim2.new(0,82,0,28),BackgroundColor3=BC,
                Text=b.Name or "OK",TextColor3=Color3.fromRGB(255,255,255),
                TextSize=12,Font=Enum.Font.GothamBold,ZIndex=43,ClipsDescendants=true,Parent=BR
            })
            Corner(B,7)
            B.MouseButton1Click:Connect(function()
                PlayClick(); Ripple(B); CloseD()
                if b.Callback then task.spawn(b.Callback) end
            end)
            B.MouseEnter:Connect(function() Tween(B,{BackgroundTransparency=0.2},0.12) end)
            B.MouseLeave:Connect(function() Tween(B,{BackgroundTransparency=0},0.12) end)
        end
    end

    function Window:Destroy()
        Tween(Win,{BackgroundTransparency=1},0.25)
        task.delay(0.3,function() Gui:Destroy() end)
    end

    return Window
