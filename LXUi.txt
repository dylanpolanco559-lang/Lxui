-- ─── Utility ─────────────────────────────────────────────────────────────────
local function Tween(obj, props, duration, style, direction)
    style     = style or Enum.EasingStyle.Quart
    direction = direction or Enum.EasingDirection.Out
    local ti  = TweenInfo.new(duration or 0.25, style, direction)
    local tw  = TweenService:Create(obj, ti, props)
    tw:Play()
    return tw
end

local function New(class, props, children)
    local obj = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then
            obj[k] = v
        end
    end
    for _, child in ipairs(children or {}) do
        child.Parent = obj
    end
    if props and props.Parent then
        obj.Parent = props.Parent
    end
    return obj
end

local function MakeDraggable(frame, handle)
    handle = handle or frame
    local dragging, dragStart, startPos = false, nil, nil
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging  = true
            dragStart = input.Position
            startPos  = frame.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

local function AddCorner(parent, radius)
    return New("UICorner", { CornerRadius = UDim.new(0, radius or 8), Parent = parent })
end

local function AddStroke(parent, color, thickness, transparency)
    return New("UIStroke", {
        Color        = color or Color3.fromRGB(255, 255, 255),
        Thickness    = thickness or 1,
        Transparency = transparency or 0.85,
        Parent       = parent
    })
end

local function AddPadding(parent, top, right, bottom, left)
    return New("UIPadding", {
        PaddingTop    = UDim.new(0, top    or 8),
        PaddingRight  = UDim.new(0, right  or 8),
        PaddingBottom = UDim.new(0, bottom or 8),
        PaddingLeft   = UDim.new(0, left   or 8),
        Parent        = parent
    })
end

local function AddListLayout(parent, padding, dir, halign, valign)
    return New("UIListLayout", {
        Padding           = UDim.new(0, padding or 6),
        FillDirection     = dir    or Enum.FillDirection.Vertical,
        HorizontalAlignment = halign or Enum.HorizontalAlignment.Left,
        VerticalAlignment   = valign or Enum.VerticalAlignment.Top,
        SortOrder         = Enum.SortOrder.LayoutOrder,
        Parent            = parent
    })
end

local function Ripple(btn, color)
    local rip = New("Frame", {
        Size            = UDim2.new(0, 0, 0, 0),
        AnchorPoint     = Vector2.new(0.5, 0.5),
        BackgroundColor3 = color or Color3.fromRGB(255,255,255),
        BackgroundTransparency = 0.6,
        BorderSizePixel = 0,
        ClipsDescendants = false,
        ZIndex          = btn.ZIndex + 1,
        Parent          = btn
    })
    AddCorner(rip, 999)
    local mp = UserInputService:GetMouseLocation()
    local bp = btn.AbsolutePosition
    rip.Position = UDim2.new(0, mp.X - bp.X, 0, mp.Y - bp.Y)
    local size = math.max(btn.AbsoluteSize.X, btn.AbsoluteSize.Y) * 2.5
    Tween(rip, { Size = UDim2.new(0, size, 0, size), BackgroundTransparency = 1 }, 0.5, Enum.EasingStyle.Quad)
    task.delay(0.5, function() rip:Destroy() end)
end

        end

                -- ─ Slider ────────────────────────────────────────────────────
                function target:AddSlider(cfg)
                    cfg = cfg or {}
                    local SlideName     = cfg.Name     or "Slider"
                    local SlideMin      = cfg.Min      or 0
                    local SlideMax      = cfg.Max      or 100
                    local SlideDefault  = cfg.Default  or SlideMin
                    local SlideDecimal  = cfg.Decimal  or 0
                    local SlideSuffix   = cfg.Suffix   or ""
                    local SlideCallback = cfg.Callback or function() end
                    local SlideFlag     = cfg.Flag

                    local SlideVal = SlideDefault

                    local SlideFrame = New("Frame", {
                        Size             = UDim2.new(1, 0, 0, 54),
                        BackgroundColor3 = Theme.BgElement,
                        ZIndex           = 7,
                        Parent           = container,
                    })
                    AddCorner(SlideFrame, 8)
                    AddStroke(SlideFrame, Theme.Border, 1, 0.7)
                    AddPadding(SlideFrame, 10, 14, 10, 14)

                    local TopRow = New("Frame", {
                        Size             = UDim2.new(1, 0, 0, 16),
                        BackgroundTransparency = 1,
                        ZIndex           = 8,
                        Parent           = SlideFrame,
                    })

                    New("TextLabel", {
                        Size             = UDim2.new(0.7, 0, 1, 0),
                        BackgroundTransparency = 1,
                        Text             = SlideName,
                        TextColor3       = Theme.Text,
                        TextSize         = 13,
                        Font             = Enum.Font.Gotham,
                        TextXAlignment   = Enum.TextXAlignment.Left,
                        ZIndex           = 9,
                        Parent           = TopRow,
                    })

                    local ValLabel = New("TextLabel", {
                        Size             = UDim2.new(0.3, 0, 1, 0),
                        Position         = UDim2.new(0.7, 0, 0, 0),
                        BackgroundTransparency = 1,
                        Text             = tostring(SlideVal) .. SlideSuffix,
                        TextColor3       = Theme.Accent,
                        TextSize         = 13,
                        Font             = Enum.Font.GothamBold,
                        TextXAlignment   = Enum.TextXAlignment.Right,
                        ZIndex           = 9,
                        Parent           = TopRow,
                    })

                    local TrackBg = New("Frame", {
                        Position         = UDim2.new(0, 0, 0, 22),
                        Size             = UDim2.new(1, 0, 0, 8),
                        BackgroundColor3 = Theme.BgThird,
                        ZIndex           = 8,
                        Parent           = SlideFrame,
                    })
                    AddCorner(TrackBg, 50)

                    local TrackFill = New("Frame", {
                        Size             = UDim2.new((SlideVal - SlideMin) / (SlideMax - SlideMin), 0, 1, 0),
                        BackgroundColor3 = Theme.Accent,
                        ZIndex           = 9,
                        Parent           = TrackBg,
                    })
                    AddCorner(TrackFill, 50)

                    local SliderKnob = New("Frame", {
                        AnchorPoint      = Vector2.new(0.5, 0.5),
                        Position         = UDim2.new((SlideVal - SlideMin) / (SlideMax - SlideMin), 0, 0.5, 0),
                        Size             = UDim2.new(0, 16, 0, 16),
                        BackgroundColor3 = Color3.fromRGB(255, 255, 255),
                        ZIndex           = 10,
                        Parent           = TrackBg,
                    })
                    AddCorner(SliderKnob, 50)
                    AddStroke(SliderKnob, Theme.Accent, 2, 0)

                    local sliding = false

                    local function SetSlider(val, silent)
                        val = math.clamp(val, SlideMin, SlideMax)
                        val = math.floor(val * (10 ^ SlideDecimal) + 0.5) / (10 ^ SlideDecimal)
                        SlideVal = val
                        local pct = (val - SlideMin) / (SlideMax - SlideMin)
                        Tween(TrackFill, { Size = UDim2.new(pct, 0, 1, 0) }, 0.1)
                        Tween(SliderKnob, { Position = UDim2.new(pct, 0, 0.5, 0) }, 0.1)
                        ValLabel.Text = string.format("%." .. SlideDecimal .. "f", val) .. SlideSuffix
                        if not silent then
                            task.spawn(SlideCallback, val)
                        end
                        if SlideFlag then
                            LXUi.Flags = LXUi.Flags or {}
                            LXUi.Flags[SlideFlag] = val
                        end
                    end

                    TrackBg.InputBegan:Connect(function(input)
                        if input.UserInputType == Enum.UserInputType.MouseButton1
                            or input.UserInputType == Enum.UserInputType.Touch then
                            sliding = true
                        end
                    end)
                    UserInputService.InputEnded:Connect(function(input)
                        if input.UserInputType == Enum.UserInputType.MouseButton1
                            or input.UserInputType == Enum.UserInputType.Touch then
                            sliding = false
                        end
                    end)
                    UserInputService.InputChanged:Connect(function(input)
                        if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement
                            or input.UserInputType == Enum.UserInputType.Touch) then
                            local rel = (input.Position.X - TrackBg.AbsolutePosition.X) / TrackBg.AbsoluteSize.X
                            local val = SlideMin + rel * (SlideMax - SlideMin)
                            SetSlider(val)
                        end
                    end)

                    SetSlider(SlideDefault, true)

                    local SlideObj = {}
                    function SlideObj:Set(val) SetSlider(val) end
                    function SlideObj:Get() return SlideVal end
                    return SlideObj
                end

                        CPValue = color
                        ColorSwatch.BackgroundColor3 = color
                        HexBox.Text = string.format("#%02X%02X%02X",
                            math.floor(color.R * 255 + 0.5),
                            math.floor(color.G * 255 + 0.5),
                            math.floor(color.B * 255 + 0.5)
                        )
                        task.spawn(CPCallback, color)

                            LXUi.Flags = LXUi.Flags or {}
                            LXUi.Flags[CPFlag] = color
                        end

                        end

                            HueKnob.Position = UDim2.new(H, 0, 0.5, 0)
                            UpdateColor(Color3.fromHSV(H, math.max(S, 0.01), math.max(V, 0.01)))
                        end

                        local hex = HexBox.Text:gsub("#", "")
                        if #hex == 6 then
                            local r = tonumber("0x"..hex:sub(1,2)) or 0
                            local g = tonumber("0x"..hex:sub(3,4)) or 0
                            local b = tonumber("0x"..hex:sub(5,6)) or 0

                            task.delay(0.2, function() PickerFrame.Visible = false end)
                        end
                    end)

                    UpdateColor(CPDefault)

                    local CPObj = {}
                    function CPObj:Set(color) UpdateColor(color) end
                    function CPObj:Get() return CPValue end

                end

                        ZIndex           = 8,

                    })

                        if Listening then return end
                        Listening    = true
                        KBBtn.Text   = "..."
                        KBBtn.TextColor3 = Theme.Warning

                        local conn
                        conn = UserInputService.InputBegan:Connect(function(input, gp)
                            if gp then return end
                            if input.UserInputType == Enum.UserInputType.Keyboard then
                                CurrentKey   = input.KeyCode
                                KBBtn.Text   = input.KeyCode.Name
                                KBBtn.TextColor3 = Theme.Accent
                                Listening    = false
                                if KBFlag then
                                    LXUi.Flags = LXUi.Flags or {}
                                    LXUi.Flags[KBFlag] = input.KeyCode
                                end
                                conn:Disconnect()
                            end
                        end)
                    end)

                    -- Key press detection
                    UserInputService.InputBegan:Connect(function(input, gp)
                        if gp then return end
                        if input.KeyCode == CurrentKey and CurrentKey ~= Enum.KeyCode.Unknown then
                            task.spawn(KBCallback, CurrentKey)
                        end
                    end)

                        CurrentKey   = keyCode
                        KBBtn.Text   = keyCode.Name
                    end

    -- ── Theme change ─────────────────────────────────────────────────────────
    function Window:SetTheme(themeName)
        -- Quickly change the accent color (full re-theme is complex; this is a shortcut)
        local newTheme = LXUi.Themes[themeName]
        if not newTheme then return end
        ActiveTheme = newTheme
        self:Notify({
            Title   = "Theme Changed",
            Content = "Theme set to " .. themeName,
            Type    = "Info",
        })
    end

    -- ── Destroy ──────────────────────────────────────────────────────────────
    function Window:Destroy()
        Tween(WindowFrame, { BackgroundTransparency = 1 }, 0.3)
        Tween(Overlay, { BackgroundTransparency = 1 }, 0.3)
        if BlurEffect then Tween(BlurEffect, { Size = 0 }, 0.3) end
        task.delay(0.35, function()
            ScreenGui:Destroy()
            if BlurEffect then BlurEffect:Destroy() end
        end)
    end

    return Window
