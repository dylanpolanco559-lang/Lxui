local function Tw(o, p, t, s, d)
    if not (o and o.Parent ~= nil or typeof(o) == "Instance") then return end
    local ok, tw = pcall(function()
        return TweenService:Create(o, TweenInfo.new(t or .22, s or Enum.EasingStyle.Quart, d or Enum.EasingDirection.Out), p)
    end)
    if ok and tw then tw:Play(); return tw end
end

local function N(cls, props)
    local o = Instance.new(cls)
    for k, v in pairs(props or {}) do if k ~= "Parent" then o[k] = v end end
    if props and props.Parent then o.Parent = props.Parent end
    return o
end

local function Rnd(b, r)  return N("UICorner", {CornerRadius=UDim.new(0, r or 8), Parent=b}) end
local function Bdr(b, c, w, t) return N("UIStroke", {Color=c or Color3.new(1,1,1), Thickness=w or 1, Transparency=t or .8, Parent=b}) end
local function Pad(b, t, r, bm, l) return N("UIPadding", {PaddingTop=UDim.new(0,t or 8), PaddingRight=UDim.new(0,r or 8), PaddingBottom=UDim.new(0,bm or 8), PaddingLeft=UDim.new(0,l or 8), Parent=b}) end
local function Lst(b, g, fd, ha, va)
    return N("UIListLayout", {
        Padding=UDim.new(0,g or 6),
        FillDirection=fd or Enum.FillDirection.Vertical,
        HorizontalAlignment=ha or Enum.HorizontalAlignment.Left,
        VerticalAlignment=va or Enum.VerticalAlignment.Top,
        SortOrder=Enum.SortOrder.LayoutOrder, Parent=b
    })
end
local function AutoY(scr, layout)
    layout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
        scr.CanvasSize = UDim2.new(0, 0, 0, layout.AbsoluteContentSize.Y + 14)
    end)
end

local function Ripple(btn, col)
    local r = N("Frame", {
        Size=UDim2.new(0,0,0,0), AnchorPoint=Vector2.new(.5,.5),
        BackgroundColor3=col or Color3.fromRGB(255,255,255),
        BackgroundTransparency=.65, ZIndex=btn.ZIndex+5, Parent=btn
    })
    Rnd(r, 999)
    local mp = UserInputService:GetMouseLocation()
    local bp = btn.AbsolutePosition
    r.Position = UDim2.new(0, mp.X-bp.X, 0, mp.Y-bp.Y)
    local sz = math.max(btn.AbsoluteSize.X, btn.AbsoluteSize.Y) * 2.6
    Tw(r, {Size=UDim2.new(0,sz,0,sz), BackgroundTransparency=1}, .45, Enum.EasingStyle.Quad)
    task.delay(.5, function() r:Destroy() end)
end

local function Drag(frame, handle)
    handle = handle or frame
    local on, ds, sp = false
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            on=true; ds=i.Position; sp=frame.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if on and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - ds
            frame.Position = UDim2.new(sp.X.Scale, sp.X.Offset+d.X, sp.Y.Scale, sp.Y.Offset+d.Y)
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then on=false end
    end)
end

local function SaveFlags(name)
    pcall(function()
        local s=""
        for k,v in pairs(LXUi.Flags) do
            local vt=typeof(v)
            if vt=="boolean" or vt=="number" or vt=="string" then
                s=s..k.."="..tostring(v).."|"..vt.."\n"
            end
        end
        writefile("LXUi_"..name..".cfg", s)
    end)
end
local function LoadFlags(name)
    pcall(function()
        if not isfile("LXUi_"..name..".cfg") then return end
        for line in readfile("LXUi_"..name..".cfg"):gmatch("[^\n]+") do
            local k,v,t=line:match("^(.-)=(.-)|(.-)\n?$")
            if k and v and t then
                if t=="boolean" then LXUi.Flags[k]=(v=="true")
                elseif t=="number" then LXUi.Flags[k]=tonumber(v) or 0
                else LXUi.Flags[k]=v end
            end
        end
    end)
end
